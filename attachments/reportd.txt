# lib/memory.py

from __future__ import annotations
import json
from dataclasses import dataclass, asdict, field
from pathlib import Path
from typing import Dict, List, Optional

DATA_DIR = Path("data")
MEMORY_STATE_PATH = DATA_DIR / "memory_state.json"


@dataclass
class MemoryItem:
    """One remembered thing.

    id:          stable key (e.g., doc_id, snippet hash)
    kind:        'doc', 'snippet', 'event', etc.
    payload:     small dict with whatever you care about
    hits:        how many times this has been retrieved
    last_ts:     last access timestamp (float, seconds)
    """

    id: str
    kind: str
    payload: Dict
    hits: int = 0
    last_ts: float = 0.0


@dataclass
class TriBufferState:
    """
    index     = long-term known things (canonical facts/docs)
    mirror    = short-term, hot items (recently used / re-queried)
    sublimate = background “cloud” (aggregated hints, counts, etc.)

    fullness_threshold controls when an item bubbles up.
    """

    index: Dict[str, MemoryItem] = field(default_factory=dict)
    mirror: Dict[str, MemoryItem] = field(default_factory=dict)
    sublimate: Dict[str, MemoryItem] = field(default_factory=dict)
    fullness_threshold: int = 3  # how many hits before it moves up

    def to_json(self) -> dict:
        def encode(d: Dict[str, MemoryItem]) -> Dict[str, dict]:
            return {k: asdict(v) for k, v in d.items()}

        return {
            "index": encode(self.index),
            "mirror": encode(self.mirror),
            "sublimate": encode(self.sublimate),
            "fullness_threshold": self.fullness_threshold,
        }

    @classmethod
    def from_json(cls, data: dict) -> "TriBufferState":
        def decode(d: Dict[str, dict]) -> Dict[str, MemoryItem]:
            return {k: MemoryItem(**v) for k, v in d.items()}

        return cls(
            index=decode(data.get("index", {})),
            mirror=decode(data.get("mirror", {})),
            sublimate=decode(data.get("sublimate", {})),
            fullness_threshold=data.get("fullness_threshold", 3),
        )


class TriBufferMemory:
    """Tri-buffer memory with very simple rules.

    - new items → sublimate
    - items used a few times → mirror
    - items used many times → index
    """

    def __init__(self, path: Path = MEMORY_STATE_PATH):
        self.path = path
        DATA_DIR.mkdir(exist_ok=True)
        if self.path.exists():
            self.state = TriBufferState.from_json(json.loads(self.path.read_text()))
        else:
            self.state = TriBufferState()
            self._save()

    def _save(self) -> None:
        self.path.write_text(json.dumps(self.state.to_json(), ensure_ascii=False, indent=2))

    # --- public API ---

    def upsert(self, item: MemoryItem, bucket: str = "sublimate") -> None:
        if bucket not in ("index", "mirror", "sublimate"):
            raise ValueError(f"Unknown bucket: {bucket}")
        getattr(self.state, bucket)[item.id] = item
        self._save()

    def record_hit(self, item_id: str, ts: float) -> Optional[MemoryItem]:
        """Increment hits + move item up if needed."""
        # search all buckets
        for bucket_name in ("index", "mirror", "sublimate"):
            bucket: Dict[str, MemoryItem] = getattr(self.state, bucket_name)
            if item_id in bucket:
                item = bucket[item_id]
                item.hits += 1
                item.last_ts = ts

                # move up if needed
                if bucket_name == "sublimate" and item.hits >= self.state.fullness_threshold:
                    # move to mirror
                    del bucket[item_id]
                    self.state.mirror[item_id] = item
                elif bucket_name == "mirror" and item.hits >= self.state.fullness_threshold * 2:
                    # move to index
                    del bucket[item_id]
                    self.state.index[item_id] = item

                self._save()
                return item
        return None

    def get(self, item_id: str) -> Optional[MemoryItem]:
        for bucket_name in ("index", "mirror", "sublimate"):
            bucket: Dict[str, MemoryItem] = getattr(self.state, bucket_name)
            if item_id in bucket:
                return bucket[item_id]
        return None

    def list_bucket(self, bucket: str) -> List[MemoryItem]:
        if bucket not in ("index", "mirror", "sublimate"):
            raise ValueError(f"Unknown bucket: {bucket}")
        return list(getattr(self.state, bucket).values())

    def snapshot(self) -> dict:
        """Return a human-friendly snapshot for debugging / UI."""
        return {
            "index_count": len(self.state.index),
            "mirror_count": len(self.state.mirror),
            "sublimate_count": len(self.state.sublimate),
            "fullness_threshold": self.state.fullness_threshold,
        }

# lib/lattice_indexer.py

from __future__ import annotations
import hashlib
import json
from dataclasses import dataclass, asdict
from pathlib import Path
from typing import List, Dict, Optional

DOCS_DIR = Path("docs")
DATA_DIR = Path("data")
INDEX_PATH = DATA_DIR / "index.jsonl"


@dataclass
class LatticeCoord:
    """Lightweight coordinate in the Möbii³us lattice.

    For now:
      face   = which domain face (e.g. 'F07xx' for social/cred)
      ring   = coarse radius / importance band
      angle  = coarse thematic angle (0-359 as int)
    """

    face: str
    ring: int
    angle: int


@dataclass
class DocIndexEntry:
    doc_id: str
    path: str
    title: str
    coord: LatticeCoord
    tags: List[str]


def _hash_path(path: Path) -> str:
    return hashlib.sha256(str(path).encode("utf-8")).hexdigest()[:12]


def _guess_face(path: Path, text: str) -> str:
    """Very dumb heuristic for now; upgrade later.

    You can wire this into real domain detection when you’re ready.
    """
    p = str(path).lower()
    t = text.lower()

    if any(k in p for k in ("cred", "social", "prison", "doinbetter")):
        return "F07xx"  # social/cred
    if any(k in p for k in ("ethic", "guardian", "noahide")):
        return "F02xx"  # ethics
    if any(k in p for k in ("flood", "hazard", "climate", "geo")):
        return "F0101"  # hazards/geo
    if any(k in p for k in ("memory", "tri-buffer", "tri_buffer")):
        return "F06xx"  # memory
    if any(k in p for k in ("story", "parable", "narrative", "comic")):
        return "F04xx"  # narrative
    # default: general/self
    return "F03xx"


def _basic_tags(text: str) -> List[str]:
    t = text.lower()
    tags = []
    for word in ("hazard", "ethics", "noahide", "cred", "garden", "prison",
                 "audio", "home", "tinybox", "aet", "gsoc", "mum", "dale"):
        if word in t:
            tags.append(word)
    return tags


def _angle_from_hash(doc_id: str) -> int:
    return int(hashlib.md5(doc_id.encode("utf-8")).hexdigest(), 16) % 360


def build_index() -> List[DocIndexEntry]:
    DATA_DIR.mkdir(exist_ok=True)
    entries: List[DocIndexEntry] = []

    for path in sorted(DOCS_DIR.rglob("*")):
        if not path.is_file():
            continue
        if path.suffix.lower() not in (".txt", ".md", ".html"):
            continue

        try:
            text = path.read_text(encoding="utf-8", errors="ignore")
        except Exception:
            continue

        doc_id = _hash_path(path)
        face = _guess_face(path, text)
        tags = _basic_tags(text)
        angle = _angle_from_hash(doc_id)
        ring = 1  # you can upgrade this to use importance later

        entry = DocIndexEntry(
            doc_id=doc_id,
            path=str(path),
            title=path.name,
            coord=LatticeCoord(face=face, ring=ring, angle=angle),
            tags=tags,
        )
        entries.append(entry)

    # write JSONL
    with INDEX_PATH.open("w", encoding="utf-8") as f:
        for e in entries:
            f.write(json.dumps(asdict(e), ensure_ascii=False) + "\n")

    return entries


def load_index() -> List[DocIndexEntry]:
    if not INDEX_PATH.exists():
        return []
    entries: List[DocIndexEntry] = []
    with INDEX_PATH.open("r", encoding="utf-8") as f:
        for line in f:
            if not line.strip():
                continue
            raw = json.loads(line)
            coord = LatticeCoord(**raw["coord"])
            raw["coord"] = coord
            entries.append(DocIndexEntry(**raw))
    return entries


def find_by_tag(tag: str) -> List[DocIndexEntry]:
    tag = tag.lower()
    return [e for e in load_index() if tag in (t.lower() for t in e.tags)]
