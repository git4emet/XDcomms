    #!/usr/bin/env python3
    \"\"\"
    Home Indexer v0.1

      - home_index.json : structured index

    import sys
    import os
    import json
    import datetime
    import html

    TEXT_EXT = {".txt", ".md", ".rst", ".log", ".html", ".htm", ".json"}
    AUDIO_EXT = {".mp3", ".wav", ".m4a", ".flac", ".ogg"}
    VIDEO_EXT = {".mp4", ".mkv", ".mov", ".avi", ".webm"}

    def human_size(n):
        for unit in ["B", "KB", "MB", "GB", "TB"]:
            if n < 1024:
                return f"{n:.1f} {unit}"
            n /= 1024
        return f"{n:.1f} PB"

    def read_snippet(path, limit=1000):
        try:
            with open(path, "r", encoding="utf-8", errors="ignore") as f:
                data = f.read(limit)
            return data
        except Exception:
            return ""

    def build_index(root):
        records = []
        root = os.path.abspath(root)
        for dirpath, dirnames, filenames in os.walk(root):
            for name in filenames:
                full = os.path.join(dirpath, name)
                rel = os.path.relpath(full, root)
                ext = os.path.splitext(name)[1].lower()

                try:
                    stat = os.stat(full)
                except OSError:
                    continue

                kind = "other"
                if ext in TEXT_EXT:
                    kind = "text"
                elif ext in AUDIO_EXT:
                    kind = "audio"
                elif ext in VIDEO_EXT:
                    kind = "video"

                rec = {
                    "path": rel.replace("\\\\", "/"),
                    "kind": kind,
                    "ext": ext,
                    "size_bytes": stat.st_size,
                    "size": human_size(stat.st_size),
                    "mtime": datetime.datetime.fromtimestamp(
                        stat.st_mtime
                    ).isoformat(timespec="seconds"),
                }

                if kind == "text":
                    snippet = read_snippet(full)
                    rec["snippet"] = snippet
                else:
                    rec["snippet"] = ""

                records.append(rec)

        return {
            "root": root,
            "generated_at": datetime.datetime.utcnow().isoformat(timespec="seconds") + "Z",
            "count": len(records),
            "items": records,
        }

from dataclasses import dataclass
from enum import Enum
from typing import Dict, Optional, List


class Lane(str, Enum):
    LITERAL = "literal"
    PARABLE = "parable"
    SYSTEM = "system"
    ETHICS = "ethics"
    SELF = "self"


class GuardianVerdict(str, Enum):
    ALLOW = "allow"
    ASK = "ask"
    BLOCK = "block"

def _base_weights_from_phase(phase: str) -> Dict[Lane, float]:
    """Phase-wise default weights before risk/memory/guardian adjustments."""
    # Sensible defaults; tune later.
    if phase in ("ingest", "spin"):
        return {
            Lane.LITERAL: 0.6,
            Lane.PARABLE: 0.3,
            Lane.SYSTEM: 0.7,
            Lane.ETHICS: 0.8,
            Lane.SELF: 0.4,
        }
    if phase in ("fold", "summit"):
        return {
            Lane.LITERAL: 0.5,
            Lane.PARABLE: 0.5,
            Lane.SYSTEM: 0.9,
            Lane.ETHICS: 0.9,
            Lane.SELF: 0.6,
        }
    if phase in ("descent", "report"):
        return {
            Lane.LITERAL: 0.7,
            Lane.PARABLE: 0.6,
            Lane.SYSTEM: 0.8,
            Lane.ETHICS: 0.8,
            Lane.SELF: 0.7,
        }
    # Fallback
    return {
        Lane.LITERAL: 0.5,
        Lane.PARABLE: 0.5,
        Lane.SYSTEM: 0.5,
        Lane.ETHICS: 0.8,
        Lane.SELF: 0.5,
    }


def _apply_risk(weights: Dict[Lane, float], risk_state: str) -> None:
    """Boost ethics/system under risk; keep parable/self lower when hot."""
    if risk_state in ("high", "critical"):
        weights[Lane.ETHICS] *= 1.5
        weights[Lane.SYSTEM] *= 1.2
        weights[Lane.PARABLE] *= 0.7
        weights[Lane.SELF] *= 0.8


def _apply_guardian(weights: Dict[Lane, float],
                    verdict: GuardianVerdict,
                    guardian_score: float,
                    constraints: List[str]) -> None:
    """Guardian modulates where we lean interpretation-wise."""
    if verdict == GuardianVerdict.BLOCK:
        constraints.append("guardian_block")
        # Ethics lane becomes dominant for explanation/redirect
        weights[Lane.ETHICS] *= 2.0
        weights[Lane.SYSTEM] *= 1.2
        weights[Lane.PARABLE] *= 0.5
        weights[Lane.SELF] *= 0.3
    elif verdict == GuardianVerdict.ASK:
        constraints.append("guardian_ask")
        # Ethics slightly favored, self toned down
        weights[Lane.ETHICS] *= 1.3
        weights[Lane.SELF] *= 0.8
    else:
        constraints.append("guardian_allow")
        # No hard clamp; guardian_score could gently shape future logic.


def _apply_memory(weights: Dict[Lane, float], memory_state: MemoryState) -> None:
    """
    If a lane has been stable over many passes (e.g., 10, 54, 101),
    gently bias toward it as “older self consistency”.
    """
    for lane, stability in memory_state.lane_stability.items():
        if stability >= 101:
            weights[lane] *= 1.4
        elif stability >= 54:
            weights[lane] *= 1.2
        elif stability >= 10:
            weights[lane] *= 1.1
    # Echo strength can give a small bump too
    for lane, echo in memory_state.echo_strength.items():
        weights[lane] *= (1.0 + 0.2 * echo)


def _normalize(weights: Dict[Lane, float]) -> Dict[Lane, float]:
    total = sum(max(v, 0.0) for v in weights.values())
    if total <= 0:
        # fallback to uniform
        n = len(weights)
        return {lane: 1.0 / n for lane in weights}
    return {lane: max(v, 0.0) / total for lane, v in weights.items()}


def unify(itop: ITOP5Result, ctx: Context) -> UnifiedInterpretation:
    """
    The Unifier Layer: pick primary lane, produce weights + unified text.
    Deterministic and explainable: given the same inputs, same output.
    """
    constraints: List[str] = []

    # 1) Start with phase defaults
    weights = _base_weights_from_phase(ctx.phase)

    # 2) Risk shaping
    _apply_risk(weights, ctx.risk_state)

    # 3) Guardian shaping
    _apply_guardian(weights, ctx.guardian_verdict, ctx.guardian_score, constraints)

    # 4) Memory resonance
    _apply_memory(weights, ctx.memory_state)

    # 5) Normalize
    weights = _normalize(weights)

    # 6) Choose primary lane (max weight)
    primary_lane = max(weights, key=weights.get)

    # 7) Unified interpretation = text from primary lane
    primary_interp = itop.lanes[primary_lane].text

    debug = {
        "raw_weights": {lane.value: float(w) for lane, w in weights.items()},
        "user_state": ctx.user_state,
        "risk_state": ctx.risk_state,
        "phase_in": ctx.phase,
        "guardian_verdict": ctx.guardian_verdict.value,
    }

    return UnifiedInterpretation(
        primary_lane=primary_lane,
        lane_weights={lane: float(w) for lane, w in weights.items()},
        interpretation=primary_interp,
        constraints=constraints,
        phase=ctx.phase,
        debug=debug,
    )
